SDK Example Gallery

Copyable, verified code recipes demonstrating real-world Limoxel SDK usage.

1. Basic Workspace & Symbol Discovery

Initialize the client, query file counts, and search symbols in a single workflow.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.OpenWorkspace(ctx, ".") defer client.Close() res, _ := client.Search().SearchSymbols(ctx, "New", sdk.PaginationOptions{Limit: 5}) for _, m := range res.Matches { fmt.Printf("%s (%s)\n", m.Name, m.Location) } }

2. CI Architecture & Health Quality Gate

Evaluate overall repository health scores and detect circular dependencies.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.OpenWorkspace(ctx, ".") defer client.Close() health, _ := client.Analysis().RepositoryHealth(ctx) fmt.Printf("Health: %.1f/100 (Grade: %s)\n", health.OverallScore, health.Grade) deps, _ := client.Analysis().AnalyzeDependencies(ctx, "") fmt.Printf("Circular Dependencies: %d\n", deps.CircularDependencies) }

3. Exporting Knowledge Graphs to Mermaid

Render relationship graphs into Mermaid flowchart format for documentation.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.OpenWorkspace(ctx, ".") defer client.Close() res, _ := client.Graph().ExportGraph(ctx, sdk.GraphFilter{MaxDepth: 2}, sdk.ExportFormatMermaid) fmt.Println(res.Content) }

4. Definition & Reference Navigation

Find symbol definitions and all incoming references across packages.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.OpenWorkspace(ctx, ".") defer client.Close() def, _ := client.Navigation().GoToDefinition(ctx, "Calculate") if def.Target != nil { fmt.Printf("Defined at: %s:%d\n", def.Target.FilePath, def.Target.Line) } }

5. Blast-Radius Change Impact Reasoning

Compute downstream impacted entities and risk score for a proposed change.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.OpenWorkspace(ctx, ".") defer client.Close() impact, _ := client.Reasoning().AnalyzeImpact(ctx, "pkg/core") fmt.Printf("Risk: %s (Impact Score: %.1f)\n", impact.RiskLevel, impact.ImpactScore) }

6. Real-Time Event Streaming

Subscribe to repository lifecycle and analysis completion events asynchronously.

package main import ( "context" "fmt" "github.com/unhield/limoxel/sdk" ) func main() { ctx := context.Background() client, _ := sdk.New(sdk.WithWorkspace(".")) defer client.Close() sub, _ := client.Events().Subscribe(ctx, sdk.EventTypeRepositoryOpened, func(ev sdk.Event) { fmt.Printf("Event: %s at %s\n", ev.Type(), ev.Timestamp()) }) defer sub.Unsubscribe() _, _ = client.Repository().Open(ctx, ".") }
