Discovery & Query

1. Workspace Initialization & Symbol Discovery

Open a local workspace session, query repository metadata, and execute symbol searches across package boundaries.

package main import ( "context" "fmt" "log" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second) defer cancel() client, err := sdk.OpenWorkspace(ctx, ".") if err != nil { log.Fatalf("failed to open workspace: %v", err) } defer client.Close() res, err := client.Search().SearchSymbols(ctx, "New", sdk.PaginationOptions{Limit: 5}) if err != nil { log.Fatalf("symbol search failed: %v", err) } fmt.Printf("Matched %d symbols:\n", res.TotalMatches) for _, m := range res.Matches { fmt.Printf(" - %s (%s)\n", m.Name, m.Location) } }
CI Quality Gate

2. CI Architecture & Health Quality Gate

Evaluate overall repository health scores and verify that circular package dependencies remain zero in automated build pipelines.

package main import ( "context" "fmt" "log" "os" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second) defer cancel() client, err := sdk.OpenWorkspace(ctx, ".") if err != nil { log.Fatalf("failed to open workspace: %v", err) } defer client.Close() health, err := client.Analysis().RepositoryHealth(ctx) if err != nil { log.Fatalf("health check failed: %v", err) } fmt.Printf("Health: %.1f/100 (Grade: %s)\n", health.OverallScore, health.Grade) deps, err := client.Analysis().AnalyzeDependencies(ctx, "") if err != nil { log.Fatalf("dependency analysis failed: %v", err) } if deps.CircularDependencies > 0 { fmt.Fprintf(os.Stderr, "FAILURE: Detected %d circular dependencies\n", deps.CircularDependencies) os.Exit(1) } fmt.Println("SUCCESS: Zero circular dependencies detected.") }
Knowledge Graph

3. Exporting Knowledge Graphs to Mermaid & DOT

Extract dependency topologies and render relationship subgraphs into Mermaid flowchart or Graphviz DOT formats for documentation.

package main import ( "context" "fmt" "log" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second) defer cancel() client, err := sdk.OpenWorkspace(ctx, ".") if err != nil { log.Fatalf("failed to open workspace: %v", err) } defer client.Close() filter := sdk.GraphFilter{MaxDepth: 2} res, err := client.Graph().ExportGraph(ctx, filter, sdk.ExportFormatMermaid) if err != nil { log.Fatalf("graph export failed: %v", err) } fmt.Println("--- Mermaid Knowledge Graph ---") fmt.Println(res.Content) }
Navigation

4. Definition & Reference Navigation

Resolve exact declaration sites for symbols and locate all call sites and incoming references across the entire codebase.

package main import ( "context" "fmt" "log" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second) defer cancel() client, err := sdk.OpenWorkspace(ctx, ".") if err != nil { log.Fatalf("failed to open workspace: %v", err) } defer client.Close() def, err := client.Navigation().GoToDefinition(ctx, "Calculate") if err != nil { log.Fatalf("navigation failed: %v", err) } if def.Target != nil { fmt.Printf("Symbol defined at: %s:%d\n", def.Target.FilePath, def.Target.Line) } }
Reasoning

5. Blast-Radius Change Impact Reasoning

Compute downstream impacted packages, affected consumers, and risk scores before committing structural refactorings.

package main import ( "context" "fmt" "log" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second) defer cancel() client, err := sdk.OpenWorkspace(ctx, ".") if err != nil { log.Fatalf("failed to open workspace: %v", err) } defer client.Close() impact, err := client.Reasoning().AnalyzeImpact(ctx, "internal/storage") if err != nil { log.Fatalf("impact reasoning failed: %v", err) } fmt.Printf("Target: internal/storage\n") fmt.Printf("Risk Level: %s (Impact Score: %.1f)\n", impact.RiskLevel, impact.ImpactScore) fmt.Printf("Directly Impacted: %d, Indirectly Impacted: %d\n", len(impact.DirectlyImpacted), len(impact.IndirectlyImpacted)) }
Event Streaming

6. Asynchronous Real-Time Event Streaming

Subscribe to repository lifecycle events and analysis milestones over thread-safe event subscriptions.

package main import ( "context" "fmt" "log" "time" "github.com/unhield/limoxel/sdk" ) func main() { ctx, cancel := context.WithCancel(context.Background()) defer cancel() client, err := sdk.New(sdk.WithWorkspace(".")) if err != nil { log.Fatalf("client init failed: %v", err) } defer client.Close() sub, err := client.Events().Subscribe(ctx, sdk.EventTypeRepositoryOpened, func(ev sdk.Event) { fmt.Printf("[EVENT RECEIVED] Type=%s Time=%s\n", ev.Type(), ev.Timestamp()) }) if err != nil { log.Fatalf("subscribe failed: %v", err) } defer sub.Unsubscribe() _, _ = client.Repository().Open(ctx, ".") time.Sleep(100 * time.Millisecond) // Allow async dispatcher flush }
